cg_bot/

├── widget.py                # Generates embeddable chat widget JavaScript
├── cli.py                   # Typer + Rich CLI (dashboard, serve, ingest, user management)
├── routers/
│   ├── __init__.py
│   ├── admin_routes.py      # Admin actions (config updates, file listing)
│   ├── analytics_routes.py  # Usage and error reporting endpoints
│   ├── auth_routes.py       # Token issuance and Azure AD token validation
│   ├── chat_routes.py       # RAG chat endpoint and feedback/history helpers
│   ├── config_routes.py     # Tenant/agent configuration CRUD
│   ├── ingest_routes.py     # File upload, delete, and ingestion triggers
│   └── user_routes.py       # User CRUD operations
├── utils/
│   ├── __init__.py
│   ├── file_processors.py   # Text extraction + OCR detection for various file types
│   ├── google_drive.py      # Drive listing and download helpers
│   └── web_scraper.py       # Sitemap parsing and page download helpers
├── static/                  # Admin HTML and widget assets served by FastAPI
├── docs/                    # Additional guides (Azure AD setup, background ingest)
├── configs/                 # Runtime tenant/agent configs (created lazily)
├── uploads/                 # Uploaded files grouped by tenant/agent
├── vector_store/            # FAISS indexes per tenant/agent with meta.json
├── users.json               # Local user database (created on first run)
├── chat_logs.db             # SQLite database (created on first run)
├── login-app/               # Sample Express API for Entra login testing
